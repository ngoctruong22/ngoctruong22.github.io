df['average_score'] = df[['math_score', 'reading_score', 'writing_score']].mean(axis=1)

# Дополнительный признак: общий уровень результата студента
def score_level(score):
    if score >= 80:
        return 'high'
    elif score >= 60:
        return 'medium'
    else:
        return 'low'

df['score_level'] = df['average_score'].apply(score_level)
